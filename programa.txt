FUNC comprobar_primo(numero)(es_primo) {
    VAR es_primo : 1; 
    VAR i : 2;

    IF (numero < 2) {
        VAR es_primo : 0;
        FINISH;
    }

    WHILE (i < numero) {
        IF (~(numero % i)) {
            VAR es_primo : 0;
            FINISH; 
        }
        VAR i : i + 1;
    }
}

FUNC calcular_potencia(base, exponente)(resultado) {
    VAR resultado : 1;
    VAR iteracion : 0;
    
    WHILE (iteracion < exponente) {
        VAR resultado : resultado * base;
        VAR iteracion : iteracion + 1;
    }
}

PRINT "Dime un numero entero positivo:";
VAR num_usuario : INPUT;

VAR resultado_primo : comprobar_primo(num_usuario);

IF (resultado_primo) {
    PRINT "Increible, es un numero primo!";
    PRINT "Vamos a elevarlo al cuadrado:";
    VAR cuadrado : calcular_potencia(num_usuario, 2);
    PRINT cuadrado;
} IFNOT {
    PRINT "Vaya, no es un numero primo.";
    
    IF (~(num_usuario % 2) & (num_usuario > 10)) {
        PRINT "Pero al menos es par y mayor que 10!";
    } IFNOT {
        PRINT "Y tampoco es par y mayor que 10 a la vez.";
    }
}
